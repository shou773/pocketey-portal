import {chromium,expect} from '@playwright/test';import {play,read} from '../tests/games/input';import {SAVE_KEY} from '../src/games/model';import fs from 'node:fs';
const out='/tmp/pocketey-ui-live-verified';fs.mkdirSync(out,{recursive:true});
const browser=await chromium.launch({executablePath:'/usr/bin/chromium',proxy:{server:process.env.HTTPS_PROXY!},args:['--no-sandbox','--use-gl=angle','--use-angle=swiftshader','--enable-unsafe-swiftshader']});const results=[];
try{for(const kind of ['orbit','amber'] as const){const c=await browser.newContext({locale:'ja-JP',viewport:{width:390,height:844},isMobile:true,hasTouch:true,deviceScaleFactor:1}),p=await c.newPage(),errors:string[]=[];p.on('pageerror',e=>errors.push(e.message));const route=`https://pocketey.com/games/${kind==='orbit'?'orbit-ribbon':'amber-step'}/`;const response=await p.goto(route,{waitUntil:'networkidle'});expect(response!.status()).toBe(200);const gl=await p.locator('canvas').evaluate(c=>{const gl=(c as HTMLCanvasElement).getContext('webgl2')!,ext=gl.getExtension('WEBGL_debug_renderer_info');return{webgl2:!!gl,renderer:ext?gl.getParameter(ext.UNMASKED_RENDERER_WEBGL):gl.getParameter(gl.RENDERER)}});
await p.getByRole('button',{name:'ステージ 1 をはじめる'}).click();await expect(p.locator('#game')).toHaveAttribute('data-mode','play');await p.locator('#pause').click();await expect(p.locator('#game')).toHaveAttribute('data-mode','pause');const frozen=await read(p);await p.waitForTimeout(250);expect((await read(p)).x).toBe(frozen.x);await p.getByRole('button',{name:'つづける'}).click();await expect(p.locator('#game')).toHaveAttribute('data-mode','play');await p.keyboard.press('KeyR');await expect(p.locator('#game')).toHaveAttribute('data-status','running');await play(p,kind,0,true);await p.locator('.hero-mark').evaluate(async e=>{await Promise.all(e.getAnimations().map(a=>a.finished));});await p.screenshot({path:`${out}/${kind}-live-clear.png`});const saved=await p.evaluate(k=>JSON.parse(localStorage.getItem(k)!),SAVE_KEY);expect(saved[kind].unlocked).toBe(2);expect(saved[kind].best[0]).toBeGreaterThan(0);await p.locator('#sound').click();await p.reload();await expect(p.locator('#sound')).toHaveText('音 ON');await expect(p.getByRole('button',{name:/ステージ 2 /})).toBeEnabled();await p.getByRole('button',{name:'ステージ 1 をはじめる'}).click();await expect(p.locator('#game')).toHaveAttribute('data-mode','play');if(kind==='amber')await p.keyboard.down('ArrowRight');await expect(p.locator('#game')).toHaveAttribute('data-status','dead',{timeout:15000});await p.keyboard.up('ArrowRight');await p.getByRole('button',{name:'すぐにリトライ'}).click();await expect(p.locator('#game')).toHaveAttribute('data-status','running');expect(Number((await read(p)).x)).toBeLessThan(10);expect(errors).toEqual([]);results.push({kind,url:p.url(),status:response!.status(),gl,pausedPosition:frozen.x,ordinaryTouchStage1Clear:true,save:saved[kind],soundAndUnlockSurvivedReload:true,deathAndInstantRetry:true,errors});await c.close();console.log(kind,'live play/pause/retry/save verified');}
fs.writeFileSync(out+'/report.json',JSON.stringify({mergeSha:'34ee2de4194180792ddee2ecc6905e634ee95c06',at:new Date().toISOString(),browser:browser.version(),proxy:'normal supported HTTPS_PROXY',coverage:'Actual public URLs. Ordinary injected multi-touch clear of stage 1 in each game, pause/resume, death/retry and reload persistence. Software-rendered cloud browser, not physical-device performance.',results},null,2));}finally{await browser.close()}
